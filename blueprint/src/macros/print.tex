% Make sure that arguments of \uses and \proves are real labels, by
% using invisible refs: LaTeX prints a warning if the label is not
% defined, but nothing is shown in the pdf file. We use LaTeX3
% programming, hence the expl3 package.
\ExplSyntaxOn
\NewDocumentCommand{\uses}{m}
 {\clist_map_inline:nn{#1}{\vphantom{\ref{##1}}}%
  \ignorespaces}
\NewDocumentCommand{\proves}{m}
 {\clist_map_inline:nn{#1}{\vphantom{\ref{##1}}}%
  \ignorespaces}
\ExplSyntaxOff
